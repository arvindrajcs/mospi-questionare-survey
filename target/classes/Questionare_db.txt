==========================user_details===============================

-----------------------------------------------------------------
-- 1. USER DETAILS TABLE
-----------------------------------------------------------------
CREATE TABLE user_details
(
    user_id              BIGINT GENERATED BY DEFAULT AS IDENTITY,
    username             VARCHAR(100) NOT NULL,
    password_hash        VARCHAR(500),
    first_name           VARCHAR(100),
    last_name            VARCHAR(100),
    email                VARCHAR(255),
    mobile_number        VARCHAR(20),
    is_active            BOOLEAN NOT NULL DEFAULT TRUE,
    password_expiry_at   TIMESTAMP WITH TIME ZONE,

    created_by           BIGINT,
    created_at           TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by           BIGINT,
    updated_at           TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_user_details
        PRIMARY KEY (user_id),
    CONSTRAINT uk_user_details_username
        UNIQUE (username),
    CONSTRAINT uk_user_details_email
        UNIQUE (email)
);

CREATE INDEX idx_user_details_is_active ON user_details(is_active);
CREATE INDEX idx_user_details_password_expiry ON user_details(password_expiry_at);


-----------------------------------------------------------------
-- 2. MASTER ROLE TABLE
-----------------------------------------------------------------
CREATE TABLE mst_role
(
    role_id             BIGINT GENERATED BY DEFAULT AS IDENTITY,
    role_name           VARCHAR(100) NOT NULL,
    role_description    VARCHAR(500),
    is_active           BOOLEAN NOT NULL DEFAULT TRUE,

    created_by          BIGINT,
    created_at          TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by          BIGINT,
    updated_at          TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_mst_role
        PRIMARY KEY (role_id),
    CONSTRAINT uk_mst_role_name
        UNIQUE (role_name)
);

CREATE INDEX idx_mst_role_is_active ON mst_role(is_active);


-----------------------------------------------------------------
-- 3. MASTER MENU TABLE (Improved with Self-Referencing parent_id)
-----------------------------------------------------------------
CREATE TABLE mst_menu
(
    menu_id       BIGINT GENERATED BY DEFAULT AS IDENTITY,
    parent_id     BIGINT, -- NULL matlab yeh root/parent menu hai; agar kisi ka ID hai toh wo uska child hai
    menu_name     VARCHAR(100) NOT NULL, -- Menu ka unique ya display naam
    menu_url      VARCHAR(500),
    icon_name     VARCHAR(100),
    label_name    VARCHAR(100) NOT NULL,
    menu_type     VARCHAR(30) NOT NULL DEFAULT 'MENU',
    menu_order    INTEGER NOT NULL DEFAULT 0,
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,

    created_by    BIGINT,
    created_at    TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by    BIGINT,
    updated_at    TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_mst_menu
        PRIMARY KEY (menu_id),
    
    -- Self-referencing FK for sub-menus
    CONSTRAINT fk_mst_menu_parent
        FOREIGN KEY (parent_id)
        REFERENCES mst_menu(menu_id)
        ON DELETE CASCADE,

    CONSTRAINT chk_mst_menu_order
        CHECK (menu_order >= 0)
);

CREATE INDEX idx_mst_menu_parent_id ON mst_menu(parent_id);
CREATE INDEX idx_mst_menu_is_active ON mst_menu(is_active);


-----------------------------------------------------------------
-- 4. MAPPING: USER <-> ROLE
-----------------------------------------------------------------
CREATE TABLE mapping_user_role
(
    user_role_mapping_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    user_id              BIGINT NOT NULL,
    role_id              BIGINT NOT NULL,
    is_active            BOOLEAN NOT NULL DEFAULT TRUE,

    created_by           BIGINT,
    created_at           TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by           BIGINT,
    updated_at           TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_mapping_user_role
        PRIMARY KEY (user_role_mapping_id),
    CONSTRAINT fk_mapping_user_role_user
        FOREIGN KEY (user_id)
        REFERENCES user_details(user_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_mapping_user_role_role
        FOREIGN KEY (role_id)
        REFERENCES mst_role(role_id)
        ON DELETE RESTRICT,
    CONSTRAINT uk_mapping_user_role
        UNIQUE (user_id, role_id)
);

-- Note: user_id par alag index ki zaroorat nahi hai kyunki UNIQUE constraint (user_id, role_id) ka pehla column user_id hai, jo automatically b-tree index create kar deta hai.
CREATE INDEX idx_mapping_user_role_role_id ON mapping_user_role(role_id);


-----------------------------------------------------------------
-- 5. MAPPING: ROLE <-> MENU (RBAC Permissions)
-----------------------------------------------------------------
CREATE TABLE mapping_role_menu
(
    role_menu_mapping_id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    role_id              BIGINT NOT NULL,
    menu_id              BIGINT NOT NULL,
    is_active            BOOLEAN NOT NULL DEFAULT TRUE,

    created_by           BIGINT,
    created_at           TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by           BIGINT,
    updated_at           TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_mapping_role_menu
        PRIMARY KEY (role_menu_mapping_id),
    CONSTRAINT fk_mapping_role_menu_role
        FOREIGN KEY (role_id)
        REFERENCES mst_role(role_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_mapping_role_menu_menu
        FOREIGN KEY (menu_id)
        REFERENCES mst_menu(menu_id)
        ON DELETE CASCADE,
    CONSTRAINT uk_mapping_role_menu
        UNIQUE (role_id, menu_id)
);

CREATE INDEX idx_mapping_role_menu_menu_id ON mapping_role_menu(menu_id);


-----------------------------------------------------------------
-- 6. AUDIT TRAIL: USER DETAILS TRAIL
-----------------------------------------------------------------
CREATE TABLE user_details_trail
(
    trail_id       BIGINT GENERATED BY DEFAULT AS IDENTITY,
    user_id        BIGINT NOT NULL,
    action_type    VARCHAR(50) NOT NULL,
    field_name     VARCHAR(100),
    old_value      JSONB,
    new_value      JSONB,
    ip_address     INET,
    user_agent     VARCHAR(1000),
    remarks        VARCHAR(1000),
    changed_by     BIGINT,
    changed_at     TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT pk_user_details_trail
        PRIMARY KEY (trail_id),
    CONSTRAINT fk_user_details_trail_user
        FOREIGN KEY (user_id)
        REFERENCES user_details(user_id)
        ON DELETE RESTRICT
);

CREATE INDEX idx_user_details_trail_user_id ON user_details_trail(user_id);

===================== 1. Survey Details Table===================================

CREATE TABLE survey_details
(
    survey_id       BIGINT GENERATED BY DEFAULT AS IDENTITY,
    title           VARCHAR(100) NOT NULL,
    survey_type_id  BIGINT NOT NULL,
    survey_code     VARCHAR(100) NOT NULL,
    division_id     BIGINT NOT NULL,
    description     VARCHAR(1000),
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    created_by      BIGINT,
    created_at      TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by      BIGINT,
    updated_at      TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_survey_details PRIMARY KEY (survey_id),
    CONSTRAINT uk_survey_code UNIQUE (survey_code),
    CONSTRAINT fk_survey_type FOREIGN KEY (survey_type_id) REFERENCES mst_survey_type (survey_type_id),
    CONSTRAINT fk_division FOREIGN KEY (division_id) REFERENCES mst_division (division_id)
);

======================= 2. Schedule Details Table=======================================

CREATE TABLE schedule_details
(
    schedule_id       BIGINT GENERATED BY DEFAULT AS IDENTITY,
    survey_id         BIGINT NOT NULL,
    schedule_type_id  BIGINT NOT NULL,
    is_active         BOOLEAN NOT NULL DEFAULT TRUE,
    created_by        BIGINT,
    created_at        TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by        BIGINT,
    updated_at        TIMESTAMP WITH TIME ZONE,

    CONSTRAINT pk_schedule_details PRIMARY KEY (schedule_id),
    CONSTRAINT fk_survey FOREIGN KEY (survey_id) REFERENCES survey_details (survey_id) ON DELETE CASCADE,
    CONSTRAINT fk_schedule_type FOREIGN KEY (schedule_type_id) REFERENCES mst_schedule_type (schedule_type_id)
);

====================== 3. Block Details Table (Added display_order for sequence)=========================

CREATE TABLE block_details
(
    block_id      BIGINT GENERATED BY DEFAULT AS IDENTITY,
    schedule_id   BIGINT NOT NULL,
    block_heading TEXT NOT NULL,
    display_order INT NOT NULL DEFAULT 1, -- Sequence maintain karne ke liye
    fields_data   JSONB,
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_by    BIGINT,
    created_at    TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by    BIGINT,
    updated_at    TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT pk_block_details PRIMARY KEY (block_id),
    CONSTRAINT fk_schedule_block FOREIGN KEY (schedule_id) REFERENCES schedule_details (schedule_id) ON DELETE CASCADE
);

============================ 4. Question Details Table (Added display_order for sequence)=================================

CREATE TABLE question_details
(
    question_id      BIGINT GENERATED BY DEFAULT AS IDENTITY,
    block_id         BIGINT NOT NULL,
    question_heading TEXT NOT NULL,
    display_order    INT NOT NULL DEFAULT 1, -- Sequence maintain karne ke liye
    fields_data      JSONB,
    is_active        BOOLEAN NOT NULL DEFAULT TRUE,
    created_by       BIGINT,
    created_at       TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by       BIGINT,
    updated_at       TIMESTAMP WITH TIME ZONE,
    
    CONSTRAINT pk_question_details PRIMARY KEY (question_id),
    CONSTRAINT fk_block_question FOREIGN KEY (block_id) REFERENCES block_details (block_id) ON DELETE CASCADE
);
